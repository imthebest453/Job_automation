@app.route("/")
def home():

    db = SessionLocal()

    try:
        jobs = JobRepository.get_all_jobs(db)
        preparing_jobs = db.query(Job).filter(
            Job.status == "Preparing"
        ).all()

        return render_template(
            "dashboard.html",
            jobs=jobs,
            preparing_jobs=preparing_jobs,
            best_match=best_match_calc(db) if jobs else 0
        )

    finally:
        db.close()


def best_match_calc(db):
    """Calculate best suitability match from profile against jobs."""
    from models.application_profile import ApplicationProfile
    profile = db.query(ApplicationProfile).first()
    if not profile or not jobs:
        return 0
    # Simple heuristic: jobs with more profile data matching
    # In a full implementation, this would use the matching service
    return 0


@app.post("/search")
def search_jobs():

    keyword = request.form.get("keyword", "").strip()

    if not keyword:
        return redirect("/")

    scraper = AdzunaScraper()

    jobs = scraper.search(keyword)

    db = SessionLocal()

    for job in jobs:
        try:
            JobRepository.add_job(db, job)
        except Exception:
            db.rollback()

    return render_template(
        "dashboard.html",
        jobs=jobs,
        keyword=keyword
    )

@app.post("/update-status/<int:job_id>")

    db = SessionLocal()

@app.route("/profile", methods=["GET", "POST"])
def profile():

    db = SessionLocal()

    profile = db.query(ApplicationProfile).first()

    if request.method == "POST":

        if not profile:
            profile = ApplicationProfile()
            db.add(profile)

        profile.full_name = request.form.get("full_name")
        profile.email = request.form.get("email")
        profile.phone = request.form.get("phone")
        profile.address = request.form.get("address")
        profile.city = request.form.get("city")
        profile.postcode = request.form.get("postcode")
        profile.skills = request.form.get("skills")
        profile.experience = request.form.get("experience")
        profile.education = request.form.get("education")

        cv = request.files.get("cv")

        if cv and cv.filename:

            filename = secure_filename(cv.filename)

            upload_folder = os.path.join(
                os.getcwd(),
                "uploads"
            )

            os.makedirs(
                upload_folder,
                exist_ok=True
            )

            cv_path = os.path.join(
                upload_folder,
                filename
            )

            cv.save(cv_path)

            profile.cv_path = cv_path

        db.commit()

        return redirect("/profile")

    return render_template(
        "profile.html",
        profile=profile
    )

@app.route("/apply/<int:job_id>")
def apply_job(job_id):

    db = SessionLocal()

    job = db.query(Job).filter(
        Job.id == job_id
    ).first()

    if not job:
        db.close()
        return "Job not found", 404

    # Check if already applied
    if job.status == "Applied":
        db.close()
        return "You have already applied to this job.", 409

    job_url = job.url

    # Mark job as "Preparing" so user can't apply twice
    job.status = "Preparing"
    db.commit()
    db.close()

    # Load profile for the automation
    from database.database import SessionLocal as NewSessionLocal
    from models.application_profile import ApplicationProfile

    prof_db = NewSessionLocal()
    try:
        profile = prof_db.query(ApplicationProfile).first()
    finally:
        prof_db.close()

    start_application(job_url, job_id=job.id, profile=profile)

    return redirect("/")
